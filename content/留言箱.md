<script>
(function () {
  var form = document.getElementById("whisper-form");
  if (!form) return;
  var result = document.getElementById("whisper-result");
  var btn = form.querySelector(".whisper-btn");

  form.addEventListener("submit", function (e) {
    e.preventDefault();
    result.className = "whisper-result";
    result.textContent = "正在送出……";
    btn.disabled = true;

    var data = new FormData(form);
    fetch("https://api.web3forms.com/submit", {
      method: "POST",
      headers: { Accept: "application/json" },
      body: data,
    })
      .then(function (res) { return res.json(); })
      .then(function (json) {
        if (json.success) {
          form.reset();
          result.className = "whisper-result ok";
          result.textContent = "收到啦，谢谢你的留言 🌱";
        } else {
          result.className = "whisper-result err";
          result.textContent = "送出失败了：" + (json.message || "请稍后再试");
          btn.disabled = false;
        }
      })
      .catch(function () {
        result.className = "whisper-result err";
        result.textContent = "网络好像出了点问题，稍后再试试？";
        btn.disabled = false;
      });
  });
})();
</script>
