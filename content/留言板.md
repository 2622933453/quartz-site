<script type="module">
import { createClient } from "https://esm.sh/@supabase/supabase-js@2"

// ↓↓↓ 把下面两个占位符换成你 Supabase 项目 Settings → API 里的值 ↓↓↓
const SUPABASE_URL = "https://xultukramoowygtsimfi.supabase.co"
const SUPABASE_ANON_KEY = "eyJhbGciOiJIUzI1NiIsInR5cCI6IkpXVCJ9.eyJpc3MiOiJzdXBhYmFzZSIsInJlZiI6Inh1bHR1a3JhbW9vd3lndHNpbWZpIiwicm9sZSI6ImFub24iLCJpYXQiOjE3ODE5MzE1MDYsImV4cCI6MjA5NzUwNzUwNn0.d_w4wvxhlMg5sdbJsZT3CyIss0n7flgRS8zJhwehMBI"
// 私密留言转发到邮箱用（复用现有 Web3Forms key）
const WEB3FORMS_KEY = "99adb30b-efa0-4935-8c5a-79e3faa094cd"

function initBoard() {
  const form = document.getElementById("board-form")
  const list = document.getElementById("board-list")
  if (!form || !list || form.dataset.bound) return
  form.dataset.bound = "1"

  const result = document.getElementById("board-result")
  const btn = form.querySelector(".board-btn")
  const textarea = document.getElementById("board-content")

  const supabase = createClient(SUPABASE_URL, SUPABASE_ANON_KEY)

  function escapeHtml(s) {
    return s.replace(/[&<>"']/g, (c) => ({
      "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;",
    }[c]))
  }

  function formatTime(iso) {
    const d = new Date(iso)
    const pad = (n) => String(n).padStart(2, "0")
    return `${d.getFullYear()}-${pad(d.getMonth() + 1)}-${pad(d.getDate())} ${pad(d.getHours())}:${pad(d.getMinutes())}`
  }

  async function loadMessages() {
    const { data, error } = await supabase
      .from("messages")
      .select("content, created_at")
      .eq("is_public", true)
      .order("created_at", { ascending: false })
      .limit(200)

    if (error) {
      list.innerHTML = '<p class="board-empty">留言加载失败了，稍后再试试。</p>'
      return
    }
    if (!data || data.length === 0) {
      list.innerHTML = '<p class="board-empty">还没有公开留言，来当第一个吧。</p>'
      return
    }
    list.innerHTML = data
      .map(
        (m) =>
          `<div class="board-item"><p class="board-item-content">${escapeHtml(
            m.content,
          )}</p><span class="board-item-time">${formatTime(m.created_at)}</span></div>`,
      )
      .join("")
  }

  form.addEventListener("submit", async (e) => {
    e.preventDefault()
    if (form.botcheck && form.botcheck.checked) return // 蜜罐命中，静默丢弃
    const content = textarea.value.trim()
    if (!content) return

    const isPublic =
      form.querySelector('input[name="visibility"]:checked').value === "public"

    result.className = "board-result"
    result.textContent = "正在送出……"
    btn.disabled = true

    const { error } = await supabase
      .from("messages")
      .insert({ content, is_public: isPublic })

    if (error) {
      result.className = "board-result err"
      result.textContent = "送出失败了，稍后再试试？"
      btn.disabled = false
      return
    }

    // 私密留言额外发一份到邮箱
    if (!isPublic) {
      try {
        await fetch("https://api.web3forms.com/submit", {
          method: "POST",
          headers: { Accept: "application/json", "Content-Type": "application/json" },
          body: JSON.stringify({
            access_key: WEB3FORMS_KEY,
            subject: "小浩日记 · 收到一条私密留言",
            from_name: "留言板",
            留言: content,
          }),
        })
      } catch (_) {}
    }

    form.reset()
    result.className = "board-result ok"
    result.textContent = isPublic
      ? "收到啦，谢谢你的留言 🌱"
      : "收到啦，这条留言是我们的秘密🌱"
    btn.disabled = false
    if (isPublic) loadMessages()
  })

  loadMessages()
}

initBoard()
document.addEventListener("nav", initBoard) // Quartz SPA 切换页面后重新初始化
</script>
