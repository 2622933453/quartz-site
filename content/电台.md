<script>
(function () {
  const WEB3FORMS_KEY = "99adb30b-efa0-4935-8c5a-79e3faa094cd"

  function initRadioLetters() {
    const toggle = document.getElementById("radio-letter-toggle")
    const form = document.getElementById("radio-letter-form")
    if (!form) return

    if (toggle && !toggle.dataset.bound) {
      toggle.dataset.bound = "1"
      toggle.addEventListener("click", () => {
        const willOpen = form.hidden
        form.hidden = !willOpen
        toggle.setAttribute("aria-expanded", String(willOpen))
        toggle.textContent = willOpen ? "收起" : "写来信"
        if (willOpen) document.getElementById("radio-name").focus()
      })
    }

    if (form.dataset.bound) return
    form.dataset.bound = "1"

    const result = document.getElementById("radio-result")
    const submit = form.querySelector(".radio-submit")
    const content = document.getElementById("radio-content")
    const name = document.getElementById("radio-name")

    form.addEventListener("submit", async (event) => {
      event.preventDefault()
      if (form.botcheck && form.botcheck.checked) return

      const letter = content.value.trim()
      const senderName = name.value.trim()
      if (!senderName || !letter) return

      result.className = "radio-result"
      result.textContent = "正在寄出……"
      submit.disabled = true

      try {
        const response = await fetch("https://api.web3forms.com/submit", {
          method: "POST",
          headers: {
            Accept: "application/json",
            "Content-Type": "application/json",
          },
          body: JSON.stringify({
            access_key: WEB3FORMS_KEY,
            subject: "小浩日记 · 电台来信",
            from_name: "电台来信箱",
            称呼: senderName,
            使用方式: "直接转发到站长邮箱，不公开显示",
            来信: letter,
          }),
        })
        const data = await response.json()
        if (!response.ok || data.success === false) {
          throw new Error("submit failed")
        }

        form.reset()
        result.className = "radio-result ok"
        result.textContent = "收到啦，来信已经寄到我这里。"
      } catch (_) {
        result.className = "radio-result err"
        result.textContent = "寄出失败了，稍后再试试？"
      } finally {
        submit.disabled = false
      }
    })
  }

  initRadioLetters()
  document.addEventListener("nav", initRadioLetters)
})()
</script>
